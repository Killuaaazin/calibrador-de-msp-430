void Set_DCO(unsigned int Delta){            // Set DCO to selected frequency
  unsigned int Compare, Oldcapture = 0;

  P1DIR &= ~BIT1;//p1.1 como entrada;
  P1SEL |= BIT1;//p1.1 funÃ§Ã£o;
  P1SEL2 &= ~BIT1;//p1.1 CCI0A; datasheet g2553 pag 43, na tabela. 0 1 0 dir sel sel2, respectivamente

  //BCSCTL1 |= DIVA_3; // ACLK = LFXT1CLK/8  //TACCTL0 = CM_1 + CCIS_1 + CAP;            // CAP, ACLK

  TA0CCTL0 = CM_3  |  // ambas as bordas
		  CCIS_0   |  // CCI0A (P1.1), para medir a largura do pulso do TX vindo do computador via C# a 1200 bps 8N1 0x55
		  SCS      |  // sincronizado
		  CAP;		  // modo captura
  TA0CTL = TASSEL_2 + MC_2 + TACLR;          // SMCLK, cont-mode, clear
  unsigned int tempo_inicial = TA0R; //to = 0;
  char e_ja_capturou_antes = 0;

  while (1)  {
	if (CCIFG & TACCTL0){		// capturou uma borda?
		TACCTL0 &= ~CCIFG;      // Capturou e limpo a flag

		if (e_ja_capturou_antes) {//fora essa atual captura, jÃ¡ capturou antes?
				Compare = TACCR0; //sim!, entÃ£o Get current captured SMCLK
				Compare = Compare - Oldcapture;//calcule o delta_t,  SMCLK difference

				if (Delta == Compare)
				break;// If equal, leave "while(1)" e sai da funÃ§Ã£o
				else if (Delta < Compare)		{
					DCOCTL--;                           // DCO is too fast, slow it down
					if (DCOCTL == 0xFF)// Did DCO roll under?
					if (BCSCTL1 & 0x0f)
					BCSCTL1--;// Select lower RSEL
				}
				else		{
					DCOCTL++;                            // DCO is too slow, speed it up
					if (DCOCTL == 0x00)// Did DCO roll over?
					if ((BCSCTL1 & 0x0f) != 0x0f)
					BCSCTL1++;// Sel higher RSEL
				}

		}//if e_ja_capturou_antes

		Oldcapture = TACCR0;// Save current captured SMCLK
		e_ja_capturou_antes = 1;
		tempo_inicial = TA0R; //to = 0;
	}//if capturou uma borda

	if ((unsigned int)(TA0R - tempo_inicial) > (unsigned int)10000){
		e_ja_capturou_antes = 0;
	}

  }//do while(1)

  TACCTL0 = 0;                              // Stop TACCR0
  TACTL = 0;                                // Stop Timer_A   BCSCTL1 &= ~DIVA_3;                       // ACLK = LFXT1CLK
}

// InicializaÃ§Ã£o UART
void UART_init(void){
    P1SEL  |= BIT1 + BIT2;  // P1.1 RXD, P1.2 TXD
    P1SEL2 |= BIT1 + BIT2;

    UCA0CTL1 |= UCSWRST; // Inicializa USCI
    UCA0CTL1 |= UCSSEL_2;   // SMCLK

    // 1MHz / 2400 = 416.67
    UCA0BR0 = 0xA0;
    UCA0BR1 = 0x01;
    UCA0MCTL = UCBRS_6;

    UCA0CTL1 &= ~UCSWRST; // Inicializa USCI

    IE2 |= UCA0RXIE; // Habilita interrupÃ§Ã£o RX
}

// Envia um caractere
void UART_sendChar(char c){
    while (!(IFG2 & UCA0TXIFG));
    UCA0TXBUF = c;
}

// Envia string
void UART_sendString(const char *str){
    while (*str) { UART_sendChar(*str++);  }
}

// InterrupÃ§Ã£o RX
#pragma vector=USCIAB0RX_VECTOR
__interrupt void USCI0RX_ISR(void){
    char recebido = UCA0RXBUF;
    if (recebido == 'r') WDTCTL = 0;

    if (recebido == 'b'){////se C# confirma que recebeu 'q' do msp, podemos transmitir os valores dos registradores
    	exibir_valores = 1;
    }

    if (recebido == 's'){////s de sair e rodar a esteira
    	exibir_valores = 2;
    	P1DIR |=  BIT6;
    	P1REN |=  BIT6 | BIT7;
    	P1OUT |= BIT7;  //pull up
    	P1OUT &= ~BIT6;

    	P2DIR |= ESTEIRA;//saida
    	P2REN &= ~ESTEIRA;//sem resistor para o driver L298
    	//uart_init(); para 9600 bps, opcional
    }

    if (recebido == 'A') {
		P1OUT |= BIT6;                             //liga o LED (P1.6)
		flag_count = 1;
	} else if (recebido == 'B') {
		P1OUT &= ~BIT6;                            //desliga o LED (P1.6)
		flag_count = 0;
		counter = 0;
	}
    if (recebido == 'D'){
        teste += 1;
    } else if (recebido == 'C')
        teste -= 1;
}

#pragma vector=TIMER1_A1_VECTOR
__interrupt void TIMER1_A1_ISR_HOOK(void) {
	switch (TA1IV) {
	case TA1IV_TACCR1:
	    reseta_temporizador(&timeout);
	    //Faz a captura da borda
	    if(!captura){
	        captura = 1;
	        overflow = 0;
	    }
	    else{
	        // Calcula a variação de tempo da captura ate o momento atual.
	        deltat = (overflow + TA1CCR1) - ti;
	        P2OUT ^= BIT0;
            captura = 0;
            overflow = 0;
            if (deltat < (DELTA_MIN))
                vel = 600;//Velocidade máxima programada caso o a variação de tempo seja MUITO pequena
            else
                calc = 1;
	    }
	    ti = TA1CCR1;
		break;
	case TA1IV_TACCR2:                          // InterrupÃ§Ã£o do CCR2
		TA1CCR2 += TEMPO_TICK;
		tick++;
		// Contador para incrementar um temporizador de 40ms na porta p2.4 para teste da função de calculo de velocidade
		if (++cont >= teste){
		    cont = 0;
		    P2OUT ^= BIT4;// Incrementa contagem de tempo base
		}
		break;
	case TA1IV_TAIFG:                          //overflow do tempo para calculo de deltat se ocorrer estouro do timer de 65536
	    P2OUT ^= BIT2;
	    overflow += 65536;
		break;
	}
}

void envia(char valor){
	char y =  valor/10; char uni = valor % 10;
	while(!(IFG2 & UCA0TXIFG));
	UCA0TXBUF = (y/10 + 48);
	while(!(IFG2 & UCA0TXIFG));
	UCA0TXBUF = (y%10 + 48);
	while(!(IFG2 & UCA0TXIFG));
	UCA0TXBUF = (uni + 48);
}
